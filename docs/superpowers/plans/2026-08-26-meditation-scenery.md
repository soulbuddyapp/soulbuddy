# Meditation Scenery Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add four relevant, licensed, offline nature scenes to `Pray and remain`, with scene-aware defaults, an edge-to-edge accessible focus view, restrained motion, and an audio-only fallback.

**Architecture:** Keep bundled-scene metadata in the existing meditation domain, render scene selection and the running focus view in two focused presentation widgets, and leave all playback inside the existing audio coordinators. `MeditationSessionSheet` remains the lifecycle coordinator: it owns setup state and stop/refrain/session timers, while the focus view owns only its decorative animation and visible countdown ticker.

**Tech Stack:** Flutter 3 / Dart 3.9, Material 3, Riverpod 3, `audioplayers`, Flutter widget/golden tests, PowerShell, FFmpeg/libwebp, SHA-256 via `crypto`.

---

- [ ] **Step 4: Create the deterministic image-preparation script**

Create `tools/images/prepare_meditation_scenery.ps1`:

```powershell
param(
    [Parameter(Mandatory = $true)]
    [string] $SourceDirectory,
    [string] $OutputDirectory = ''
)

$ErrorActionPreference = 'Stop'
$repoRoot = (Resolve-Path (Join-Path $PSScriptRoot '..\..')).Path
if ([string]::IsNullOrWhiteSpace($OutputDirectory)) {
    $OutputDirectory = Join-Path $repoRoot 'apps\soulbuddy\assets\images\meditation'
}

$ffmpeg = Get-Command ffmpeg -ErrorAction Stop
$ffprobe = Get-Command ffprobe -ErrorAction Stop
$sourceRoot = (Resolve-Path -LiteralPath $SourceDirectory).Path
New-Item -ItemType Directory -Force -Path $OutputDirectory | Out-Null

$scenes = @(
    [pscustomobject]@{
        Key = 'dawn_lake'
        Source = 'dawn_lake_source.jpg'
        Width = 5901
        Height = 3841
        Filter = 'crop=ih*3/4:ih:(iw-ih*3/4)/2:0,scale=1440:1920:flags=lanczos'
        Page = 'https://pixabay.com/photos/sunrise-lake-nature-dawn-water-7353034/'
        Contributor = 'Purgin_Alexandr'
        Published = '2022-07-31'
    }
    [pscustomobject]@{
        Key = 'flowing_waterfall'
        Source = 'flowing_waterfall_source.jpg'
        Width = 3174
        Height = 4791
        Filter = 'crop=iw:iw*4/3:0:0,scale=1440:1920:flags=lanczos'
        Page = 'https://pixabay.com/photos/forest-waterfall-water-nature-869716/'
        Contributor = 'Derek8978'
        Published = '2015-08-04'
    }
    [pscustomobject]@{
        Key = 'forest_path'
        Source = 'forest_path_source.jpg'
        Width = 5228
        Height = 3477
        Filter = 'crop=ih*3/4:ih:(iw-ih*3/4)/2:0,scale=1440:1920:flags=lanczos'
        Page = 'https://pixabay.com/photos/forest-path-in-the-morning-trees-5094362/'
        Contributor = 'LPArt'
        Published = '2020-04-27'
    }
    [pscustomobject]@{
        Key = 'mountain_meadow'
        Source = 'mountain_meadow_source.jpg'
        Width = 3000
        Height = 4000
        Filter = 'scale=1440:1920:flags=lanczos'
        Page = 'https://pixabay.com/photos/meadow-lake-mountains-nature-6766789/'
        Contributor = 'Aldo810'
        Published = '2021-11-04'
    }
)

foreach ($scene in $scenes) {
    $sourcePath = Join-Path $sourceRoot $scene.Source
    if (-not (Test-Path -LiteralPath $sourcePath -PathType Leaf)) {
        throw "Missing approved source: $sourcePath"
    }
    $probeJson = & $ffprobe.Source -v error -select_streams v:0 `
        -show_entries stream=width,height -of json -- $sourcePath
    $probe = ($probeJson | ConvertFrom-Json).streams[0]
    if ($probe.width -ne $scene.Width -or $probe.height -ne $scene.Height) {
        throw "Unexpected dimensions for $($scene.Source): $($probe.width)x$($probe.height)"
    }

    $outputPath = Join-Path $OutputDirectory "$($scene.Key).webp"
    foreach ($quality in 82, 78, 74, 70) {
        & $ffmpeg.Source -hide_banner -loglevel error -y -i $sourcePath `
            -vf $scene.Filter -frames:v 1 -c:v libwebp -quality $quality `
            -compression_level 6 -preset picture -map_metadata -1 $outputPath
        if ($LASTEXITCODE -ne 0) {
            throw "FFmpeg failed for $($scene.Key)"
        }
        if ((Get-Item -LiteralPath $outputPath).Length -le 409600) {
            break
        }
    }
    if ((Get-Item -LiteralPath $outputPath).Length -gt 409600) {
        throw "$($scene.Key).webp exceeds 400 KiB"
    }
}

$totalBytes = ($scenes | ForEach-Object {
    (Get-Item -LiteralPath (Join-Path $OutputDirectory "$($_.Key).webp")).Length
} | Measure-Object -Sum).Sum
if ($totalBytes -gt 1638400) {
    throw "Meditation scenery bundle exceeds 1.6 MB"
}

$manifest = @(
    '# SoulBuddy meditation scenery assets',
    '',
    'Retrieved: 2026-08-26',
    '',
    'All four source pages mark their photographs free for use under the Pixabay Content License:',
    'https://pixabay.com/service/license-summary/',
    '',
    'The files are cropped, resized, metadata-stripped, compressed, and integrated behind SoulBuddy prayer UI. They are not distributed as standalone stock files.',
    ''
)
foreach ($scene in $scenes) {
    $outputPath = Join-Path $OutputDirectory "$($scene.Key).webp"
    $hash = (Get-FileHash -Algorithm SHA256 -LiteralPath $outputPath).Hash.ToLowerInvariant()
    $bytes = (Get-Item -LiteralPath $outputPath).Length
    $manifest += @(
        "## ``$($scene.Key).webp``",
        '',
        "- Source page: $($scene.Page)",
        "- Contributor: $($scene.Contributor)",
        "- Published: $($scene.Published)",
        '- License: Pixabay Content License',
        "- Original: ``$($scene.Source)`` ($($scene.Width) x $($scene.Height) JPG)",
        "- Transform: ``$($scene.Filter)``; libwebp quality chosen from 82, 78, 74, 70 to remain at or below 400 KiB; metadata removed",
        "- Final: 1440 x 1920 WebP; $bytes bytes",
        "- SHA-256: ``$hash``",
        ''
    )
}
$manifest | Set-Content -LiteralPath (Join-Path $OutputDirectory 'ASSETS.md') -Encoding utf8

Write-Host "Prepared $($scenes.Count) scenes ($totalBytes bytes) in $OutputDirectory"
```

class _SceneryChoiceState extends State<_SceneryChoice> {
  var _reportedFailure = false;

  @override
  Widget build(BuildContext context) {
    final enabled = widget.available && !_reportedFailure;
    return Semantics(
      button: true,
      selected: widget.selected,
      enabled: enabled,
      label: enabled
          ? widget.scene.label
          : '${widget.scene.label}, Image unavailable',
      child: Material(
        color: Theme.of(context).colorScheme.surfaceContainerLow,
        shape: RoundedRectangleBorder(

class _MeditationFocusViewState extends State<MeditationFocusView>
    with SingleTickerProviderStateMixin {

  Timer? _countdown;

  late Duration _remaining;

  var _reportedImageFailure = false;

  @override
  void initState() {
    super.initState();
    _remaining = widget.duration;
    _endsAt = DateTime.now().add(widget.duration);
    _motion = AnimationController(
      vsync: this,
      duration: widget.timed ? widget.duration : const Duration(minutes: 10),
    );
    if (widget.timed) {
      _countdown = Timer.periodic(const Duration(seconds: 1), (_) {
        final next = _endsAt.difference(DateTime.now());
        if (!mounted) return;
        setState(() {
          _remaining = next.isNegative ? Duration.zero : next;
        });
      });
    }
  }

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();
    if (MediaQuery.maybeOf(context)?.disableAnimations ?? false) {
      _motion.stop();
      _motion.value = 0.5;
    } else if (!_motion.isAnimating && _motion.value < 1) {
      _motion.forward();
    }
  }

  @override
  void dispose() {
    _countdown?.cancel();

    super.dispose();

        MediaQuery.maybeOf(context)?.disableAnimations ?? false;

                    Semantics(
                      label: '${_spoken(_remaining)} remaining',
                      child: Text(
                        _clock(_remaining),
                        textAlign: TextAlign.center,
                        style: Theme.of(context).textTheme.labelLarge?.copyWith(
                          color: widget.scenery == null || _imageFailed
                              ? Theme.of(context).colorScheme.onSurface
                              : Colors.white,
                        ),
                      ),
                    ),
                  const Spacer(),
                  Center(
                    child: ConstrainedBox(
                      constraints: const BoxConstraints(maxWidth: 620),
                      child: Semantics(
                        liveRegion: widget.prompt != null,
                        header: true,
                        child: Text(

                          textAlign: TextAlign.center,

  Widget _background(bool reduceMotion) {
    final scenery = widget.scenery;
    if (scenery == null || _imageFailed) return const SizedBox.expand();
    final provider =
        widget.imageProviderBuilder?.call(scenery) ?? AssetImage(scenery.assetPath);
    final image = Semantics(
      image: true,
      label: 'Meditation scenery: ${scenery.label.toLowerCase()}',
      excludeSemantics: true,
      child: Image(
        image: provider,
        fit: BoxFit.cover,
        alignment: _alignment(scenery.preferredAlignment),
        errorBuilder: (context, error, stackTrace) {
          _reportImageFailure();
          return const SizedBox.expand();
        },
      ),
    );
    if (reduceMotion) {
      return Transform.scale(
        key: const Key('meditation-scenery-static'),
        scale: 1.025,
        child: image,
      );
    }
    return AnimatedBuilder(
      key: const Key('meditation-scenery-motion'),
      animation: _motion,
      child: image,
      builder: (context, child) => Transform.translate(
        offset: Offset(0, -MediaQuery.sizeOf(context).height * 0.02 * _motion.value),
        child: Transform.scale(
          scale: 1 + 0.05 * _motion.value,
          child: child,
        ),
      ),
    );

  void _reportImageFailure() {
    if (_reportedImageFailure) return;
    _reportedImageFailure = true;
    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (mounted) setState(() => _imageFailed = true);
    });
  }

Alignment _alignment(MeditationSceneryAlignment value) => switch (value) {
  MeditationSceneryAlignment.center => Alignment.center,
  MeditationSceneryAlignment.topCenter => Alignment.topCenter,
};

String _clock(Duration value) {
  final minutes = value.inMinutes.toString().padLeft(2, '0');
  final seconds = (value.inSeconds % 60).toString().padLeft(2, '0');
  return '$minutes:$seconds';
}

String _spoken(Duration value) =>
    '${value.inMinutes} minutes ${value.inSeconds % 60} seconds';

  void dispose() {
    _sessionTimer?.cancel();
    _repeatTimer?.cancel();
    if (_running && !_stopping) unawaited(widget.onStop?.call());
    super.dispose();
  }

  Future<void> _stop() async {
    if (!_running || _stopping) return;
    _stopping = true;
    _sessionTimer?.cancel();
    _repeatTimer?.cancel();
    _sessionTimer = null;
    _repeatTimer = null;
    try {
      await widget.onStop?.call();

        setState(() {
          _running = false;
          _stopping = false;

      }
    }
  }
